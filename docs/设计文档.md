
## 五、Worktree 管理核心实现

```typescript
// worktree-manager.ts
import { execa } from 'execa';
import * as path from 'path';
import * as fs from 'fs-extra';

export class WorktreeManager {
  private baseRepoPath: string;
  private worktreeBaseDir: string;

  constructor(repoPath: string) {
    this.baseRepoPath = repoPath;
    this.worktreeBaseDir = path.join(repoPath, '.agent-worktrees');
  }

  async createWorktree(sessionId: string, taskName: string, baseBranch = 'main') {
    const branchName = `agent/${taskName}-${sessionId.slice(0, 8)}`;
    const worktreePath = path.join(this.worktreeBaseDir, sessionId);

    // 1. 创建新分支并 checkout 到独立 worktree
    await execa('git', [
      'worktree', 'add',
      '-b', branchName,
      worktreePath,
      baseBranch
    ], { cwd: this.baseRepoPath });

    // 2. 安装依赖（如果需要）
    await this.setupWorktreeEnv(worktreePath);

    return { branchName, worktreePath };
  }

  async getDiff(sessionId: string): Promise<GitDiff> {
    const worktreePath = this.getWorktreePath(sessionId);
    
    // 获取与 base branch 的差异
    const { stdout: statOutput } = await execa('git', [
      'diff', '--stat', 'HEAD'
    ], { cwd: worktreePath });

    const { stdout: diffOutput } = await execa('git', [
      'diff', 'HEAD'
    ], { cwd: worktreePath });

    const { stdout: filesOutput } = await execa('git', [
      'diff', '--name-status', 'HEAD'
    ], { cwd: worktreePath });

    return {
      stat: statOutput,
      fullDiff: diffOutput,
      files: this.parseFileStatus(filesOutput),
      summary: this.parseStat(statOutput)
    };
  }

  async watchDiff(sessionId: string, callback: (diff: GitDiff) => void) {
    // 使用 chokidar 监听文件变化，实时推送 diff
    const worktreePath = this.getWorktreePath(sessionId);
    const watcher = chokidar.watch(worktreePath, {
      ignored: /node_modules|\.git/,
      persistent: true,
      debounce: 500
    });

    watcher.on('change', async () => {
      const diff = await this.getDiff(sessionId);
      callback(diff);
    });

    return watcher;
  }

  async mergeToBase(sessionId: string, strategy: 'squash' | 'merge' | 'rebase' = 'squash') {
    const { branchName } = await this.getSessionInfo(sessionId);
    
    // 切换到主仓库执行合并
    if (strategy === 'squash') {
      await execa('git', ['merge', '--squash', branchName], { 
        cwd: this.baseRepoPath 
      });
    }
    // ... 其他策略
  }

  async removeWorktree(sessionId: string) {
    const worktreePath = this.getWorktreePath(sessionId);
    await execa('git', ['worktree', 'remove', '--force', worktreePath], {
      cwd: this.baseRepoPath
    });
  }

  private getWorktreePath(sessionId: string) {
    return path.join(this.worktreeBaseDir, sessionId);
  }

  private async setupWorktreeEnv(worktreePath: string) {
    // 软链 node_modules 避免重复安装
    const srcModules = path.join(this.baseRepoPath, 'node_modules');
    const dstModules = path.join(worktreePath, 'node_modules');
    if (await fs.pathExists(srcModules)) {
      await fs.symlink(srcModules, dstModules);
    }
  }
}
```

## 六、终端多路复用

```typescript
// terminal-multiplexer.ts
import { IPty, spawn } from 'node-pty';
import { EventEmitter } from 'events';

export class TerminalMultiplexer extends EventEmitter {
  private terminals = new Map<string, TerminalSession>();

  createTerminal(sessionId: string, cwd: string): TerminalSession {
    const pty = spawn('bash', [], {
      name: 'xterm-256color',
      cols: 120,
      rows: 40,
      cwd,
      env: {
        ...process.env,
        AGENT_SESSION_ID: sessionId,  // 注入环境变量
      }
    });

    const session: TerminalSession = {
      id: sessionId,
      pty,
      buffer: [],
      status: 'idle'
    };

    // 捕获输出并广播
    pty.onData((data) => {
      session.buffer.push({ type: 'output', data, timestamp: Date.now() });
      this.emit('terminal:data', { sessionId, data });
      
      // 检测 Agent 的特殊标记（审批请求）
      this.detectApprovalRequest(sessionId, data);
    });

    pty.onExit(({ exitCode }) => {
      this.emit('terminal:exit', { sessionId, exitCode });
    });

    this.terminals.set(sessionId, session);
    return session;
  }

  sendToTerminal(sessionId: string, data: string) {
    const session = this.terminals.get(sessionId);
    if (session) {
      session.pty.write(data);
    }
  }

  broadcastToAll(data: string, sessionIds?: string[]) {
    const targets = sessionIds || Array.from(this.terminals.keys());
    targets.forEach(id => this.sendToTerminal(id, data));
  }

  // 检测 Agent 输出中的特殊标记，触发审批流程
  private detectApprovalRequest(sessionId: string, data: string) {
    // 约定：Agent 输出特定格式触发审批
    // e.g., [APPROVAL_REQUIRED] type=destructive command="rm -rf dist"
    const approvalPattern = /\[APPROVAL_REQUIRED\] (.+)/;
    const match = data.match(approvalPattern);
    
    if (match) {
      const approvalData = this.parseApprovalRequest(match[1]);
      this.emit('approval:required', { sessionId, ...approvalData });
    }

    // 检测 checkpoint
    const checkpointPattern = /\[CHECKPOINT\] (.+)/;
    const cpMatch = data.match(checkpointPattern);
    if (cpMatch) {
      this.emit('checkpoint:reached', { sessionId, message: cpMatch[1] });
    }
  }

  getBuffer(sessionId: string, lastN = 100) {
    return this.terminals.get(sessionId)?.buffer.slice(-lastN) || [];
  }
}
```

## 七、审批工作流

```typescript
// approval-workflow.ts
export class ApprovalWorkflow {
  private pendingApprovals = new Map<string, ApprovalRequest>();
  
  async requestApproval(
    sessionId: string, 
    request: ApprovalRequest
  ): Promise<'approved' | 'rejected'> {
    
    // 挂起 Agent 执行（暂停终端输入）
    this.terminalMux.sendToTerminal(sessionId, '');  // 发送空，等待
    
    // 更新会话状态
    await this.sessionManager.updateStatus(sessionId, 'waiting');
    
    // 推送到前端
    this.wsServer.broadcast('approval:required', {
      sessionId,
      request,
      diff: await this.worktreeManager.getDiff(sessionId)
    });

    // 返回 Promise，等待用户决策
    return new Promise((resolve) => {
      this.pendingApprovals.set(sessionId, { ...request, resolve });
    });
  }

  async handleApproval(sessionId: string, decision: 'approved' | 'rejected') {
    const pending = this.pendingApprovals.get(sessionId);
    if (!pending) return;

    if (decision === 'approved') {
      // 恢复 Agent 执行
      this.terminalMux.sendToTerminal(sessionId, 'y\n');
      await this.sessionManager.updateStatus(sessionId, 'running');
    } else {
      this.terminalMux.sendToTerminal(sessionId, 'n\n');
      await this.sessionManager.updateStatus(sessionId, 'paused');
    }

    pending.resolve(decision);
    this.pendingApprovals.delete(sessionId);
  }

  // 批量审批
  async batchApprove(sessionIds: string[]) {
    await Promise.all(sessionIds.map(id => this.handleApproval(id, 'approved')));
  }
}
```

```typescript
    // 注入 checkpoint 指令到 system prompt
    const systemPrompt = `
      当你完成重要步骤时，输出: [CHECKPOINT] <描述>
      当你需要执行危险操作时，输出: [APPROVAL_REQUIRED] type=destructive command="<命令>"
      当你准备好合并时，输出: [APPROVAL_REQUIRED] type=merge-ready
    `;
    
    this.process = spawn('claude', [
      '--system', systemPrompt,
      '--message', task,
      '--cwd', cwd
    ]);
  }
}

// Aider Adapter  
class AiderAdapter implements AgentAdapter {
  async start(task: string, cwd: string) {
    this.process = spawn('aider', [
      '--message', task,
      '--yes-always',  // 配合我们的审批系统
      '--no-auto-commits'  // 我们自己控制提交
    ], { cwd });
  }
}
```

## 十、关键交互流程

```
用户创建任务
     │
     ▼
创建 Git Worktree ──→ 独立分支隔离
     │
     ▼
启动 Agent 进程 ──→ 注入 Checkpoint 协议
     │
     ▼
实时监控输出 ──→ 推送到画布卡片
     │
  ┌──┴──┐
  │检测到│
  │审批点│
  └──┬──┘
     │
     ▼
暂停 Agent ──→ 推送审批请求到前端
     │
     ▼
用户 Review Diff ──→ 批准/拒绝
     │
  ┌──┴──┐
  │批准  │
  └──┬──┘
     │
     ▼
恢复 Agent 继续执行
     │
     ▼
完成 ──→ Squash Merge 到主分支 ──→ 清理 Worktree
```

---


这个方案的核心优势是**物理隔离**（worktree）+ **协议约定**（checkpoint标记）+ **统一指挥**（画布+看板），让你真正像指挥官一样管理多个并行 Agent。
